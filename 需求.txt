```c
static void test_straight(float dist) {
    if (g_ctrl.mode != CTRL_MODE_IDLE) {
        wireless_uart_send_string("[TEST] Controller busy\r\n");
        return;
    }
    float start_x = position.x_m;
    float target_x = start_x + dist;
    // 构建两点直线路径（注意：位置使用当前 y 坐标）
    g_ctrl.current_path[0][0] = start_x;
    g_ctrl.current_path[0][1] = position.y_m;
    g_ctrl.current_path[1][0] = target_x;
    g_ctrl.current_path[1][1] = position.y_m;
    g_ctrl.path_len = 2;
    g_ctrl.path_following = 1;
    g_ctrl.path_purpose = PATH_PURPOSE_MOVE_ACTION;
    g_ctrl.use_tangent_heading = 0;
    g_ctrl.path_locked_yaw = 0.0f;
    g_ctrl.max_speed = 0.15f;
    g_ctrl.min_speed = 0.08f;
    g_ctrl.path_tolerance = 0.03f;
    g_ctrl.push_smoothed_speed = 0.0f;
    g_ctrl.axial_tracking = 0;
    PID_Reset(&angle_trace_param);
    g_ctrl.mode = CTRL_MODE_PATH_FOLLOWING;

    while (g_ctrl.mode != CTRL_MODE_IDLE) rt_thread_mdelay(10);
    float actual = position.x_m - start_x;
    char buf[128];
    rt_sprintf(buf, "[STRAIGHT] target=%.3f, actual=%.3f, error=%.3f\r\n", dist, actual, actual - dist);
    wireless_uart_send_string(buf);
}
```

## 增加方形路径测试功能

在 `debug.c` 中添加以下内容，实现方形路径导航测试。小车将从当前位置开始，沿边长 `side` 米的正方形路径移动一圈，回到起点附近，用于测试路径跟踪精度。

### 1. 在 `debug.c` 中添加函数声明和实现

```c
// 前向声明
static void test_square(float side);

// 实现方形路径测试
static void test_square(float side) {
    if (g_ctrl.mode != CTRL_MODE_IDLE) {
        wireless_uart_send_string("[SQUARE] Controller busy\r\n");
        return;
    }
    if (side <= 0.0f) {
        wireless_uart_send_string("[SQUARE] Invalid side length\r\n");
        return;
    }

    // 起点：小车当前位置
    float start_x = position.x_m;
    float start_y = position.y_m;
    // 计算四个角点（逆时针方向，起点为左下角）
    float corner1_x = start_x;
    float corner1_y = start_y;
    float corner2_x = start_x + side;
    float corner2_y = start_y;
    float corner3_x = start_x + side;
    float corner3_y = start_y + side;
    float corner4_x = start_x;
    float corner4_y = start_y + side;
    // 终点回到起点
    float corner5_x = start_x;
    float corner5_y = start_y;

    // 插值步长（例如 0.1m 一个点）
    float step = 0.1f;
    int total_points = 0;
    float path[MAX_PATH_POINTS][2];

    // 定义添加路径点函数（避免重复代码）
    #define ADD_POINT(x, y) do { \
        if (total_points < MAX_PATH_POINTS) { \
            path[total_points][0] = (x); \
            path[total_points][1] = (y); \
            total_points++; \
        } \
    } while(0)

    // 从起点到角点2（右）
    float dx = corner2_x - corner1_x;
    float dy = corner2_y - corner1_y;
    float len = sqrtf(dx*dx + dy*dy);
    int steps = (int)(len / step);
    if (steps < 1) steps = 1;
    for (int i = 0; i <= steps; i++) {
        float fx = corner1_x + (dx * i / steps);
        float fy = corner1_y + (dy * i / steps);
        ADD_POINT(fx, fy);
    }
    // 从角点2到角点3（上）
    dx = corner3_x - corner2_x;
    dy = corner3_y - corner2_y;
    len = sqrtf(dx*dx + dy*dy);
    steps = (int)(len / step);
    if (steps < 1) steps = 1;
    for (int i = 1; i <= steps; i++) {
        float fx = corner2_x + (dx * i / steps);
        float fy = corner2_y + (dy * i / steps);
        ADD_POINT(fx, fy);
    }
    // 从角点3到角点4（左）
    dx = corner4_x - corner3_x;
    dy = corner4_y - corner3_y;
    len = sqrtf(dx*dx + dy*dy);
    steps = (int)(len / step);
    if (steps < 1) steps = 1;
    for (int i = 1; i <= steps; i++) {
        float fx = corner3_x + (dx * i / steps);
        float fy = corner3_y + (dy * i / steps);
        ADD_POINT(fx, fy);
    }
    // 从角点4到起点（下）
    dx = corner5_x - corner4_x;
    dy = corner5_y - corner4_y;
    len = sqrtf(dx*dx + dy*dy);
    steps = (int)(len / step);
    if (steps < 1) steps = 1;
    for (int i = 1; i <= steps; i++) {
        float fx = corner4_x + (dx * i / steps);
        float fy = corner4_y + (dy * i / steps);
        ADD_POINT(fx, fy);
    }

    if (total_points < 2) {
        wireless_uart_send_string("[SQUARE] Not enough path points\r\n");
        return;
    }

    // 将路径复制到控制器
    for (int i = 0; i < total_points; i++) {
        g_ctrl.current_path[i][0] = path[i][0];
        g_ctrl.current_path[i][1] = path[i][1];
    }
    g_ctrl.path_len = total_points;
    g_ctrl.path_following = 1;
    g_ctrl.path_purpose = PATH_PURPOSE_MOVE_ACTION;
    g_ctrl.use_tangent_heading = 0;   // 锁航向0°
    g_ctrl.path_locked_yaw = 0.0f;
    g_ctrl.push_smoothed_speed = 0.0f;
    g_ctrl.axial_tracking = 0;        // 方形路径可能包含斜线？不，这里是轴向移动，但为了通用设为0
    // 可选：调整速度和容差（使用当前参数）
    PID_Reset(&angle_trace_param);
    g_ctrl.mode = CTRL_MODE_PATH_FOLLOWING;

    wireless_uart_send_string("[SQUARE] Start square path...\r\n");
    while (g_ctrl.mode != CTRL_MODE_IDLE) {
        rt_thread_mdelay(10);
    }

    // 计算终点误差（起点和终点位置差）
    float end_x = position.x_m;
    float end_y = position.y_m;
    float dx_end = end_x - start_x;
    float dy_end = end_y - start_y;
    float dist_err = sqrtf(dx_end*dx_end + dy_end*dy_end);
    char buf[128];
    rt_sprintf(buf, "[SQUARE] Finished. Start(%.3f,%.3f) End(%.3f,%.3f) Error=%.3f m\r\n",
               start_x, start_y, end_x, end_y, dist_err);
    wireless_uart_send_string(buf);
}
```

### 3. 使用示例

- 先加载空地图：`test empty`
- 然后运行方形测试：`test square 1.0`  
  小车将沿着一个边长 1 米的正方形走一圈，最后打印起点和终点的误差。

### 4. 注意事项

- 方形路径的起点是当前小车位置，因此建议在开始测试前将小车放置在地图中心或指定位置（可通过 `test nav` 移动到目标点）。
- 如果希望起点固定为地图中心，可以在 `test square` 中先调用 `Position_Set(地图中心x, 地图中心y, 0)`，但地图中心坐标取决于地图尺寸（例如地图宽 3.2m，高 2.4m，中心为 (1.6, 1.2)）。也可以增加一个命令 `set_pose 1.6 1.2` 来手动设置。
- 路径插值步长设为 0.1m，可保证路径点足够密集，跟踪平滑。如需调整，可修改 `step` 变量。

此测试能有效评估路径跟踪算法在直角转弯时的性能。
